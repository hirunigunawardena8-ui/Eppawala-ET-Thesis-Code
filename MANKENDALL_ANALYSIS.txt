// 2) CLOUD MASK FUNCTION
function maskL8L2(img) {

  var qa = img.select('QA_PIXEL');

  var mask = qa.bitwiseAnd(1 << 3).eq(0)
    .and(qa.bitwiseAnd(1 << 4).eq(0))
    .and(qa.bitwiseAnd(1 << 0).eq(0));

  var sr = img.select([
      'SR_B2',
      'SR_B3',
      'SR_B4',
      'SR_B5',
      'SR_B6',
      'SR_B7'
    ])
    .multiply(0.0000275)
    .add(-0.2);

  return img
    .addBands(sr, null, true)
    .updateMask(mask)
    .copyProperties(img, ['system:time_start']);
}

// 3) SURFACE PARAMETERS

function addSurfaceParams(image) {

  // NDVI
  var ndvi = image.normalizedDifference(
    ['SR_B5', 'SR_B4']
  ).rename('NDVI');

  // ALBEDO
  var albedo = image.expression(
    '-0.0015 + 0.160*B2 + 0.291*B3 + 0.243*B4 + 0.116*B5 + 0.112*B6 + 0.081*B7',
    {
      'B2': image.select('SR_B2'),
      'B3': image.select('SR_B3'),
      'B4': image.select('SR_B4'),
      'B5': image.select('SR_B5'),
      'B6': image.select('SR_B6'),
      'B7': image.select('SR_B7')
    }
  ).rename('Albedo');

  // LAND SURFACE TEMPERATURE
  var lst = image.select('ST_B10')
    .multiply(0.00341802)
    .add(149.0)
    .subtract(273.15)
    .rename('LST');

  return image.addBands([
    ndvi,
    albedo,
    lst
  ]);
}

// 4) ERA5 RADIATION

function getERA5Radiation(start, end) {

  var era5 = ee.ImageCollection(
      'ECMWF/ERA5_LAND/DAILY_AGGR'
    )
    .filterDate(start, end)
    .select(
      'surface_solar_radiation_downwards_sum'
    );

  var Rs = era5.mean()
    .divide(86400)
    .rename('Rs');

  return Rs;
}

// 5) NET RADIATION

function addRn(image, Rs) {

  var emissivity = image.select('NDVI')
    .multiply(0.004)
    .add(0.986);

  var Ts = image.select('LST').add(273.15);

  var sigma = ee.Image.constant(5.67e-8);

  var Rl_up = Ts.pow(4)
    .multiply(sigma)
    .multiply(emissivity);

  var Rl_down = Ts.pow(4)
    .multiply(sigma)
    .multiply(0.85);

  var Rn = Rs
    .multiply(
      ee.Image.constant(1)
      .subtract(image.select('Albedo'))
    )
    .add(Rl_down)
    .subtract(Rl_up)
    .rename('Rn');

  Rn = Rn.max(0);

  return image.addBands(Rn);
}

// 6) SOIL HEAT FLUX

function addG(image) {

  var G = image.expression(

    'Rn * (0.05 + 0.18 * exp(-0.521 * NDVI))',

    {
      'Rn': image.select('Rn'),
      'NDVI': image.select('NDVI')
    }

  ).rename('G');

  G = G.max(0);

  return image.addBands(G);
}

// 7) SENSIBLE HEAT FLUX
function addH(image) {

  var lst = image.select('LST');

  var stats = lst.reduceRegion({

    reducer: ee.Reducer.percentile([5,95]),

    geometry: studyArea,

    scale: 60,

    maxPixels: 1e13

  });

  var LSTmin = ee.Number(
    ee.Algorithms.If(
      stats.get('LST_p5'),
      stats.get('LST_p5'),
      20
    )
  );

  var LSTmax = ee.Number(
    ee.Algorithms.If(
      stats.get('LST_p95'),
      stats.get('LST_p95'),
      40
    )
  );

  var dT = lst.subtract(LSTmin)
    .divide(LSTmax.subtract(LSTmin))
    .clamp(0,1);

  var H = image.expression(
  'Rn * (0.7 - 0.5 * NDVI)',
  {
    'Rn': image.select('Rn'),
    'NDVI': image.select('NDVI')
  }
).rename('H');

H = H.max(0);

  return image.addBands(H);
}

// 8) EVAPOTRANSPIRATION

function addET(image) {

  var Rn = image.select('Rn');

  var G = image.select('G');

  var H = image.select('H');

  var availableEnergy = Rn.subtract(G);

  availableEnergy = availableEnergy.max(0);

  var LE = availableEnergy.subtract(H);

  LE = LE.max(0);

  var EF = LE.divide(
    availableEnergy.max(1)
  ).clamp(0,1);

  var ET_daily = EF
    .multiply(availableEnergy)
    .multiply(0.035)
    .clamp(0,8)
    .rename('ET_daily');

  return image.addBands(ET_daily);
}

var years = ee.List.sequence(2013,2025);

var annualTable = ee.FeatureCollection(

years.map(function(y){

  y = ee.Number(y);

  var start = ee.Date.fromYMD(y,1,1);

  var end = ee.Date.fromYMD(
    y.add(1),1,1
  );

  var annualRs = getERA5Radiation(
    start,
    end
  );

  var collection = ee.ImageCollection(
      'LANDSAT/LC08/C02/T1_L2'
    )
    .filterBounds(studyArea)
    .filterDate(start,end)
    .filter(
      ee.Filter.lt(
        'CLOUD_COVER',
        50
      )
    )
    .map(maskL8L2)
    .map(addSurfaceParams)
    .map(function(img){
      return addRn(
        img,
        annualRs
      );
    })
    .map(addG)
    .map(addH)
    .map(addET);

  var annualET = collection
    .select('ET_daily')
    .mean()
    .multiply(365)
    .rename('Annual_ET');

  return annualET
    .reduceRegions({

      collection: patches,

      reducer: ee.Reducer.mean(),

      scale:30

    })

    .map(function(f){

      return f.set({

        Year:y,

        Annual_ET:f.get('mean'),

        PatchID:f.get('PatchID'),

        Zone:f.get('Zone'),

        Distance:f.get('Distance')

      });

    });

})

).flatten();

Export.table.toDrive({
  collection: annualTable,
  description:'Annual_ET_2013_2025',
  folder:'ET_Results',
  fileFormat:'CSV'
});
