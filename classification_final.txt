function maskL8(image){

  var qa = image.select('QA_PIXEL');

  var mask = qa.bitwiseAnd(1 << 3).eq(0)
      .and(qa.bitwiseAnd(1 << 4).eq(0));

  var optical = image.select([
    'SR_B2',
    'SR_B3',
    'SR_B4',
    'SR_B5',
    'SR_B6',
    'SR_B7'
  ])
  .multiply(0.0000275)
  .add(-0.2);

  return image
    .addBands(optical,null,true)
    .updateMask(mask);
}

// FUNCTION TO GET 5 PATCHES

function getPatches(region,name){

  var region3857 =
      region.transform('EPSG:3857',1);

  var grid = region3857.coveringGrid({
      proj: ee.Projection('EPSG:3857'),
      scale:500
  })
  .map(function(f){
      return f.transform('EPSG:4326',1);
  });

  var cropAreaImage =
      cropMask.multiply(ee.Image.pixelArea());

  var totalAreaImage =
      ee.Image.pixelArea();

  var gridStats = grid.map(function(cell){

      var cropArea =
          cropAreaImage.reduceRegion({
            reducer:ee.Reducer.sum(),
            geometry:cell.geometry(),
            scale:30,
            maxPixels:1e13
          });

      var totalArea =
          totalAreaImage.reduceRegion({
            reducer:ee.Reducer.sum(),
            geometry:cell.geometry(),
            scale:30,
            maxPixels:1e13
          });

      var cropPercent =
          ee.Number(cropArea.values().get(0))
          .divide(
            ee.Number(totalArea.values().get(0))
          )
          .multiply(100);

      return cell.set(
        'cropPercent',
        cropPercent
      );

  });

  var cropDominant =
      gridStats.filter(
      ee.Filter.gte(
      'cropPercent',
      75
      ));

  var selected =
      cropDominant
      .randomColumn('rand',42)
      .sort('rand')
      .limit(5);

  Map.addLayer(
      selected.style({
      color:'blue',
      fillColor:'0000FF44',
      width:2
      }),
      {},
      name
  );

  print(name,selected);

  return selected;
}
